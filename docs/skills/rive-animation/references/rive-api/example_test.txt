For scripting, you can also write tests as follows (in a separate script/module):
function setup(test: Tester)
  local group = test.group
  local case = test.case
  group('values are good', function()
    case('first test', function(expect)
      expect(2 + 2).is(4)
    end)
    case('second test', function(expect)
      expect(2).is(2)
    end)
    group('group 2', function()
      case('inner test', function(expect)
        expect(2 + 2).is(4)
      end)
      case('inner test 2', function(expect)
        expect(2 + 2).is(4)
      end)
      case('check values are matching2', function(expect)
        expect(3).lessThanOrEqual(3)
        expect('hello').is('hello')
        expect(2 + 2).is(4)
        expect(3).greaterThan(2)
        expect(4).never.greaterThanOrEqual(6)
        expect(3).lessThanOrEqual(3)
      end)
      case('check something else', function(expect)
        expect(2 + 2).is(4)
      end)
    end)
  end)
  case('ungrouped test', function(expect)
    expect(2 + 2).never.is(3)
  end)
end

return function(): Tests
  return setup
end
